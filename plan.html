Gen360 — CBA Implementation Plan

Phased migration from TOTP MFA to Certificate-Based Authentication

Project Summary

This plan details the phased implementation of Certificate-Based Authentication (CBA) for the Gen360 admin panel, replacing the existing TOTP-based MFA with a phishing-resistant, certificate-driven authentication flow.

Current State (TOTP)

  • Email + password → JWT (HS256)
  • TOTP 6-digit codes (30s window)
  • Vulnerable to phishing & real-time relay
  • Requires authenticator app
  • Custom auth library (no external deps)

Target State (CBA Hybrid)

  • X.509 client certificate → JWT
  • Browser auto-selects certificate
  • Phishing-immune (private key never leaves device)
  • No app needed — cert in OS store
  • TOTP kept as emergency fallback

Recommended Approach

🏆 Option E (Hybrid) with Option A (App-Level mTLS) as the base
Add CBA alongside existing TOTP auth. Zero disruption, zero cost, works locally, full control. Admins choose "Sign in with Certificate" or fall back to password+TOTP.

Why This Approach?

Step-by-Step Tasks

1 Create directory structure

mkdir -p certs/ca certs/clients certs/server certs/crl

# Add to .gitignore
echo "certs/ca/*.key" >> .gitignore
echo "certs/clients/*.key" >> .gitignore
echo "certs/clients/*.p12" >> .gitignore
echo "certs/server/*.key" >> .gitignore

2 Generate Root CA

# Generate 4096-bit RSA key for Root CA
openssl genrsa -aes256 -out certs/ca/root-ca.key 4096
# You will be prompted for a passphrase — store it securely

# Create self-signed Root CA certificate (valid 10 years)
openssl req -new -x509 -key certs/ca/root-ca.key \
  -out certs/ca/root-ca.pem -days 3650 \
  -subj "/C=BR/ST=DF/O=Gen360/CN=Gen360 Root CA"

# Verify the certificate
openssl x509 -in certs/ca/root-ca.pem -text -noout | head -20

3 Generate Intermediate CA

# Generate intermediate CA key
openssl genrsa -aes256 -out certs/ca/intermediate-ca.key 4096

# Create Certificate Signing Request (CSR)
openssl req -new -key certs/ca/intermediate-ca.key \
  -out certs/ca/intermediate-ca.csr \
  -subj "/C=BR/ST=DF/O=Gen360/CN=Gen360 Intermediate CA"

# Sign intermediate CA with Root CA (valid 5 years)
openssl x509 -req -in certs/ca/intermediate-ca.csr \
  -CA certs/ca/root-ca.pem -CAkey certs/ca/root-ca.key \
  -CAcreateserial -out certs/ca/intermediate-ca.pem \
  -days 1825 -sha256

# Create CA chain bundle
cat certs/ca/intermediate-ca.pem certs/ca/root-ca.pem > certs/ca/ca-chain.pem

# Verify chain
openssl verify -CAfile certs/ca/root-ca.pem certs/ca/intermediate-ca.pem

4 Generate Admin Client Certificate

# Generate client key (no passphrase for convenience, or add -aes256)
openssl genrsa -out certs/clients/admin.key 2048

# Create client CSR with admin identity
openssl req -new -key certs/clients/admin.key \
  -out certs/clients/admin.csr \
  -subj "/C=BR/ST=DF/O=Gen360/CN=Thiago Beier/emailAddress=thiago.beier@gmail.com"

# Create extension file for client auth
cat > certs/clients/admin.ext <<EOF
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage=digitalSignature,keyEncipherment
extendedKeyUsage=clientAuth
subjectAltName=email:thiago.beier@gmail.com
EOF

# Sign with Intermediate CA (valid 1 year)
openssl x509 -req -in certs/clients/admin.csr \
  -CA certs/ca/intermediate-ca.pem -CAkey certs/ca/intermediate-ca.key \
  -CAcreateserial -out certs/clients/admin.pem \
  -days 365 -sha256 -extfile certs/clients/admin.ext

# Package as PKCS#12 for browser import
openssl pkcs12 -export -out certs/clients/admin.p12 \
  -inkey certs/clients/admin.key -in certs/clients/admin.pem \
  -certfile certs/ca/ca-chain.pem \
  -name "Gen360 Admin - Thiago Beier"
# You will be prompted for an export password

# Get certificate thumbprint (needed for DB mapping)
openssl x509 -in certs/clients/admin.pem -fingerprint -sha256 -noout
# SHA256 Fingerprint=AB:CD:EF:12:34:... (save this for Phase 2)

5 Generate Server Certificate (for local HTTPS)

# Generate server key
openssl genrsa -out certs/server/server.key 2048

# Create server cert (signed by intermediate CA, includes localhost SANs)
cat > certs/server/server.ext <<EOF
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage=digitalSignature,keyEncipherment
extendedKeyUsage=serverAuth

openssl req -new -key certs/server/server.key \
  -out certs/server/server.csr \

openssl x509 -req -in certs/server/server.csr \
  -CA certs/ca/intermediate-ca.pem -CAkey certs/ca/intermediate-ca.key \

# Add these imports at the top of api/auth.py
from cryptography import x509
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.x509.oid import NameOID
import base64

def validar_certificado(cert_pem: str, ca_cert_path: str) -> dict:
    """
    Validate a client certificate against the trusted CA.
    
    Args:
        cert_pem: PEM-encoded client certificate string
        ca_cert_path: Path to CA chain PEM file
    
    Returns:
        dict with 'valid', 'subject', 'email', 'thumbprint', 'expires'
    
    Raises:
        ValueError: If certificate is invalid, expired, or untrusted
    """
    # Parse client certificate
    cert = x509.load_pem_x509_certificate(cert_pem.encode())
    
    # Check expiration
    from datetime import datetime, timezone
    now = datetime.now(timezone.utc)
    if cert.not_valid_after_utc < now:
        raise ValueError("Certificate has expired")
    if cert.not_valid_before_utc > now:
        raise ValueError("Certificate is not yet valid")
    
    # Compute SHA-256 thumbprint
    thumbprint = cert.fingerprint(hashes.SHA256()).hex()
    
    # Extract identity
    subject = cert.subject.rfc4514_string()
    email = None
    try:
        email = cert.subject.get_attributes_for_oid(
            NameOID.EMAIL_ADDRESS
        )[0].value
    except (IndexError, Exception):
        # Try SAN extension
        try:
            san = cert.extensions.get_extension_for_class(
                x509.SubjectAlternativeName
            )
            emails = san.value.get_values_for_type(x509.RFC822Name)
            if emails:
                email = emails[0]
        except x509.ExtensionNotFound:
            pass
    
    return {
        "valid": True,
        "subject": subject,
        "email": email,
        "thumbprint": thumbprint,
        "expires": cert.not_valid_after_utc.isoformat()
    }


def extrair_identidade_cert(cert_pem: str) -> dict:
    """
    Extract identity fields from a PEM certificate without validation.
    Used for the X-ARR-ClientCert header in Azure App Service.
    
    Args:
        cert_pem: PEM or Base64-encoded DER certificate
    
    Returns:
        dict with 'subject', 'email', 'thumbprint', 'cn'
    """
    # Handle Azure's Base64-encoded DER format
    if not cert_pem.startswith("-----BEGIN"):
        der_bytes = base64.b64decode(cert_pem)
        cert = x509.load_der_x509_certificate(der_bytes)
    else:
        cert = x509.load_pem_x509_certificate(cert_pem.encode())
    
    thumbprint = cert.fingerprint(hashes.SHA256()).hex()
    subject = cert.subject.rfc4514_string()
    
    cn = None
    try:
        cn = cert.subject.get_attributes_for_oid(
            NameOID.COMMON_NAME
        )[0].value
    except (IndexError, Exception):
        pass
    
    email = None
    try:
        email = cert.subject.get_attributes_for_oid(
            NameOID.EMAIL_ADDRESS
        )[0].value
    except (IndexError, Exception):
        pass
    
    return {
        "subject": subject,
        "email": email,
        "cn": cn,
        "thumbprint": thumbprint
    }

5 New database queries — api/queries.py

def obter_admin_por_cert_thumbprint(conn, thumbprint: str):
    """Look up admin by certificate SHA-256 thumbprint."""
    with conn.cursor() as cur:
        cur.execute(
            "SELECT * FROM admins WHERE cert_thumbprint = %s",
            (thumbprint,)
        )
        return cur.fetchone()


def atualizar_admin_cert(conn, admin_id: str, thumbprint: str, subject: str):
    """Bind a certificate to an admin account."""
    with conn.cursor() as cur:
        cur.execute(
            """UPDATE admins 
               SET cert_thumbprint = %s, cert_subject = %s,
                   auth_method = 'both'
               WHERE id = %s""",
            (thumbprint, subject, admin_id)
        )
        conn.commit()


def registrar_audit_log(conn, admin_id, method, success, ip, ua, cert_subj=None):
    """Record an authentication event in the audit log."""
    with conn.cursor() as cur:
        cur.execute(
            """INSERT INTO auth_audit_log 
               (admin_id, auth_method, success, ip_address, user_agent, cert_subject)
               VALUES (%s, %s, %s, %s, %s, %s)""",
            (admin_id, method, success, ip, ua, cert_subj)
        )
        conn.commit()

6 New endpoint — api/main.py

from fastapi import Request, HTTPException

@app.post("/admin/login/cert")
async def login_with_certificate(request: Request):
    """
    Authenticate admin using client certificate.
    
    Local dev:  Certificate from mTLS handshake (request.scope['transport'])
    Azure prod: Certificate from X-ARR-ClientCert header
    """
    cert_pem = None
    
    # Try Azure App Service header first
    arr_cert = request.headers.get("X-ARR-ClientCert")
    if arr_cert:
        cert_pem = arr_cert  # Base64-encoded DER
    else:
        # Try mTLS connection (local dev with Uvicorn SSL)
        transport = request.scope.get("transport")
        if transport:
            ssl_object = transport.get_extra_info("ssl_object")
            if ssl_object:
                peer_cert_der = ssl_object.getpeercert(binary_form=True)
                if peer_cert_der:
                    cert_pem = base64.b64encode(peer_cert_der).decode()
    
    if not cert_pem:
        raise HTTPException(
            status_code=401,
            detail="No client certificate provided"
        )
    
    # Extract identity from certificate
    try:
        identity = extrair_identidade_cert(cert_pem)
    except Exception as e:
        raise HTTPException(status_code=401, detail=f"Invalid certificate: {e}")
    
    # Look up admin by thumbprint
    conn = get_connection()
    try:
        admin = obter_admin_por_cert_thumbprint(conn, identity["thumbprint"])
        if not admin:
            registrar_audit_log(conn, None, "certificate", False,
                request.client.host, request.headers.get("user-agent"),
                identity["subject"])
            raise HTTPException(
                status_code=403,
                detail="Certificate not registered to any admin"
            )
        
        # Check auth method is allowed
        if admin["auth_method"] == "totp":
            raise HTTPException(
                status_code=403,
                detail="Certificate auth not enabled for this admin"
            )
        
        # Issue JWT (same as existing TOTP flow)
        token = gerar_jwt({"sub": str(admin["id"]), "email": admin["email"]})
        
        # Audit log
        registrar_audit_log(conn, admin["id"], "certificate", True,
            request.client.host, request.headers.get("user-agent"),
            identity["subject"])
        
        return {
            "access_token": token,
            "token_type": "bearer",
            "auth_method": "certificate",
            "admin_name": admin["nome"]
        }
    finally:
        conn.close()

    if (!response.ok) {
      const data = await response.json();
      throw new Error(data.detail || 'Certificate authentication failed');
    }
    
    const data = await response.json();
    
    // Store JWT (same as existing TOTP flow)
    localStorage.setItem('token', data.access_token);
    localStorage.setItem('auth_method', 'certificate');
    
    // Redirect to admin home
    window.location.href = 'admin_home.html';
    
  } catch (err) {
    errEl.textContent = err.message;
    errEl.style.display = 'block';
    
    // If cert auth fails, suggest TOTP fallback
    if (err.message.includes('No client certificate')) {
      errEl.textContent = 
        'No certificate detected. Please use password + TOTP login, '
        + 'or ensure your certificate is installed in the browser.';
    }
  } finally {
    btn.disabled = false;
    btn.innerHTML = `<svg width="20" height="20" viewBox="0 0 24 24" fill="none"
      stroke="currentColor" stroke-width="2">
      <rect x="3" y="11" width="18" height="11" rx="2" ry="2"/>
      <path d="M7 11V7a5 5 0 0 1 10 0v4"/>
    </svg> Sign in with Certificate`;
  }
}

async function bindCertificate() {
  try {
    const response = await fetch(`${API_BASE}/admin/cert/bind`, {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` },
      credentials: 'include'
    });
    if (response.ok) {
      alert('Certificate bound successfully!');
      loadCertificateStatus();
    } else {
      const err = await response.json();
      alert(`Failed: ${err.detail}`);
    }
  } catch (e) {
    alert('Error: Ensure you are connected via HTTPS and your certificate is installed.');
  }
}

# Add to api/main.py — health check enhancement

@app.get("/health/certs")
async def health_check_certs():
    """Check certificate expiration status."""
    import os
    from datetime import datetime, timezone, timedelta
    from cryptography import x509
    
    warnings = []
    ca_path = os.environ.get("CA_CERT_PATH", "certs/ca/ca-chain.pem")
    
    if os.path.exists(ca_path):
        with open(ca_path, "rb") as f:
            ca_cert = x509.load_pem_x509_certificate(f.read())
        
        days_left = (ca_cert.not_valid_after_utc - datetime.now(timezone.utc)).days
        if days_left < 30:
            warnings.append(f"CA certificate expires in {days_left} days!")
    
    # Check admin cert expiry via database
    conn = get_connection()
    try:
        with conn.cursor() as cur:
            cur.execute("SELECT email, cert_subject FROM admins WHERE cert_thumbprint IS NOT NULL")
            admins_with_certs = cur.fetchall()
    finally:
        conn.close()
    
    return {
        "status": "warning" if warnings else "healthy",
        "warnings": warnings,
        "admins_with_certs": len(admins_with_certs)
    }
