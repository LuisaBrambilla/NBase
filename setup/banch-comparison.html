3a. admin_frontend/config.js — NEW Centralized API Configuration

High Impact All Pages

A new file that auto-detects the environment and sets the correct API URL. Replaces all scattered inline <script> blocks and hard-coded URLs.

  • Localhost / file:// → http://localhost:8000
  • Azure Static Web App (*.web.core.windows.net) → https://app-gen360-api.azurewebsites.net
  • Any other host → https://app-gen360-api.azurewebsites.net

Why it matters: Without this, the app breaks when deployed to Azure because the frontend would try to call /api or localhost:8000 instead of the real API. 

+  (function () {
+    const DEFAULT_REMOTE_API = "https://app-gen360-api.azurewebsites.net";
+    const host = window.location.hostname || "";
+    const protocol = window.location.protocol || "";
+    const isLocalHost = ["localhost", "127.0.0.1", "::1"].includes(host);
+    const isFileProtocol = protocol === "file:";
+    const isAzureStatic = /\.web\.core\.windows\.net$/i.test(host);
+
+    let configured = (window.API_BASE_URL || "").trim();
+
+    if (!configured) {
+      configured = isLocalHost || isFileProtocol
+        ? "http://localhost:8000"
+        : DEFAULT_REMOTE_API;
+    }
+
+    if (configured.startsWith("/")) {
+      if (isLocalHost && window.location.origin) {
+        configured = `${window.location.origin}${configured}`;
+      } else {
+        configured = DEFAULT_REMOTE_API;
+      }
+    }
+
+    if (isAzureStatic && !isLocalHost) {
+      configured = DEFAULT_REMOTE_API;
+    }
+
+    window.API_BASE_URL = configured.replace(/\/+$/, "");
+  })();
