Task 1.3 — Generate Root CA (10-Year Validity)

⚠️ Security: The Root CA private key is the most sensitive file. Never commit it to Git. Store it offline after initial setup.
# Generate 4096-bit RSA private key for Root CA
openssl genrsa -aes256 -out certs/ca/root-ca.key 4096
# You will be prompted for a passphrase — choose a strong one and save it securely
# Enter PEM pass phrase: ********

# Generate self-signed Root CA certificate (10 years = 3650 days)
openssl req -new -x509 -key certs/ca/root-ca.key -sha256 -days 3650 ^
  -out certs/ca/root-ca.pem ^
  -subj "/C=BR/ST=DF/L=Brasilia/O=Gen360/OU=Security/CN=Gen360 Root CA"
# Enter pass phrase for certs/ca/root-ca.key: ********

# Verify the Root CA certificate
openssl x509 -text -noout -in certs/ca/root-ca.pem
# Certificate:
#     Data:
#         Version: 3 (0x2)
#         Serial Number: ...
#         Issuer: C=BR, ST=DF, L=Brasilia, O=Gen360, OU=Security, CN=Gen360 Root CA
#         Validity
#             Not Before: Apr  1 00:00:00 2026 GMT
#             Not After : Mar 29 00:00:00 2036 GMT

Task 1.4 — Generate Intermediate CA (5-Year Validity)

# Generate Intermediate CA private key (no passphrase for automation; add -aes256 if desired)
openssl genrsa -out certs/intermediate/intermediate-ca.key 4096

# Create a Certificate Signing Request (CSR)
openssl req -new -key certs/intermediate/intermediate-ca.key -sha256 ^
  -out certs/intermediate/intermediate-ca.csr ^
  -subj "/C=BR/ST=DF/L=Brasilia/O=Gen360/OU=Security/CN=Gen360 Intermediate CA"

# Create v3 extensions file for intermediate CA
echo [v3_intermediate_ca] > certs/intermediate/v3_intermediate.ext
echo subjectKeyIdentifier = hash >> certs/intermediate/v3_intermediate.ext
echo authorityKeyIdentifier = keyid:always,issuer >> certs/intermediate/v3_intermediate.ext
echo basicConstraints = critical, CA:true, pathlen:0 >> certs/intermediate/v3_intermediate.ext
echo keyUsage = critical, digitalSignature, cRLSign, keyCertSign >> certs/intermediate/v3_intermediate.ext

# Sign the Intermediate CA CSR with the Root CA (5 years = 1825 days)
openssl x509 -req -in certs/intermediate/intermediate-ca.csr ^
  -CA certs/ca/root-ca.pem -CAkey certs/ca/root-ca.key -CAcreateserial ^
  -out certs/intermediate/intermediate-ca.pem -days 1825 -sha256 ^
  -extfile certs/intermediate/v3_intermediate.ext -extensions v3_intermediate_ca
# Enter pass phrase for certs/ca/root-ca.key: ********

# Verify the Intermediate CA
openssl x509 -text -noout -in certs/intermediate/intermediate-ca.pem
openssl verify -CAfile certs/ca/root-ca.pem certs/intermediate/intermediate-ca.pem
# certs/intermediate/intermediate-ca.pem: OK

Task 1.5 — Create CA Chain Bundle

# Concatenate Intermediate + Root into a single chain file
# Order matters: leaf-most CA first, root last
type certs\intermediate\intermediate-ca.pem certs\ca\root-ca.pem > certs\ca-chain.pem

# Bash/Git Bash equivalent:
cat certs/intermediate/intermediate-ca.pem certs/ca/root-ca.pem > certs/ca-chain.pem

# Verify the chain
openssl verify -CAfile certs/ca/root-ca.pem -untrusted certs/intermediate/intermediate-ca.pem certs/ca-chain.pem

# Create v3 extensions for server cert with SAN
echo [v3_server] > certs/server/v3_server.ext
echo subjectKeyIdentifier = hash >> certs/server/v3_server.ext
echo authorityKeyIdentifier = keyid,issuer >> certs/server/v3_server.ext

Task 1.8 — Client Cert Generation Script

Create scripts/gen-client-cert.ps1:

# scripts/gen-client-cert.ps1
# Usage: .\scripts\gen-client-cert.ps1 -Email "user@example.com" -Name "Full Name"
param(
    [Parameter(Mandatory=$true)][string]$Email,
    [Parameter(Mandatory=$true)][string]$Name,
    [int]$ValidDays = 365,
    [string]$OutDir = "certs/clients"
)

$ErrorActionPreference = "Stop"
$safeName = ($Email -replace '[^a-zA-Z0-9]', '_')
$keyFile  = "$OutDir/$safeName.key"
$csrFile  = "$OutDir/$safeName.csr"
$certFile = "$OutDir/$safeName.pem"
$p12File  = "$OutDir/$safeName.p12"
$extFile  = "$OutDir/$safeName.ext"

Write-Host "Generating client certificate for: $Name <$Email>" -ForegroundColor Cyan

# 1. Generate private key
openssl genrsa -out $keyFile 2048
Write-Host "  Key:  $keyFile" -ForegroundColor Green

# 2. Create v3 extensions with email SAN
@"
[v3_client]
subjectKeyIdentifier = hash
authorityKeyIdentifier = keyid,issuer
basicConstraints = CA:FALSE
keyUsage = critical, digitalSignature, keyEncipherment
extendedKeyUsage = clientAuth
subjectAltName = email:$Email
"@ | Set-Content $extFile -Encoding ASCII

# 3. Create CSR
openssl req -new -key $keyFile -sha256 -out $csrFile `
  -subj "/C=BR/O=Gen360/CN=$Name/emailAddress=$Email"
Write-Host "  CSR:  $csrFile" -ForegroundColor Green

# 4. Sign with Intermediate CA
openssl x509 -req -in $csrFile `
  -CA certs/intermediate/intermediate-ca.pem `
  -CAkey certs/intermediate/intermediate-ca.key `
  -CAcreateserial -out $certFile -days $ValidDays -sha256 `
  -extfile $extFile -extensions v3_client
Write-Host "  Cert: $certFile" -ForegroundColor Green

# 5. Package as PKCS#12 (.p12) for browser import
openssl pkcs12 -export -out $p12File `
  -inkey $keyFile -in $certFile `
  -certfile certs/ca-chain.pem `
  -name "$Name ($Email)"
Write-Host "  P12:  $p12File (import this into your browser)" -ForegroundColor Green

# 6. Show thumbprint
$thumb = openssl x509 -fingerprint -sha256 -noout -in $certFile
Write-Host "  Thumbprint: $thumb" -ForegroundColor Yellow

# Cleanup temp files
Remove-Item $csrFile, $extFile -ErrorAction SilentlyContinue

Write-Host "`nDone! Import $p12File into your browser or OS certificate store." -ForegroundColor Cyan

-- Migration: Add certificate-based authentication fields to admins table
-- Date: 2026-04-01
-- Description: Adds columns for storing client certificate thumbprints,
--   certificate subject (DN), and preferred auth method (totp|cert|both).

-- 1. Add certificate thumbprint column (SHA-256, 64 hex chars)
ALTER TABLE admins
  ADD COLUMN IF NOT EXISTS cert_thumbprint VARCHAR(64) DEFAULT NULL;

-- 2. Add certificate subject (Distinguished Name from cert)
ALTER TABLE admins
  ADD COLUMN IF NOT EXISTS cert_subject VARCHAR(512) DEFAULT NULL;

-- 3. Add auth method preference
--    'totp'  = TOTP MFA only (current behavior)
--    'cert'  = Certificate only
--    'both'  = Both TOTP + cert (highest security)
ALTER TABLE admins
  ADD COLUMN IF NOT EXISTS auth_method VARCHAR(10) DEFAULT 'totp';

-- 4. Add certificate registration date
ALTER TABLE admins
  ADD COLUMN IF NOT EXISTS cert_registered_at TIMESTAMP DEFAULT NULL;

-- 5. Create unique index on cert_thumbprint for fast lookups
CREATE UNIQUE INDEX IF NOT EXISTS idx_admins_cert_thumbprint
  ON admins(cert_thumbprint)
  WHERE cert_thumbprint IS NOT NULL;

-- 6. Add CHECK constraint for valid auth_method values
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'chk_admin_auth_method'
  ) THEN
    ALTER TABLE admins
      ADD CONSTRAINT chk_admin_auth_method
      CHECK (auth_method IN ('totp', 'cert', 'both'));
  END IF;
END
$$;

-- 7. Add audit trigger for cert changes
CREATE OR REPLACE FUNCTION audit_cert_change() RETURNS trigger AS $$
BEGIN
  IF (OLD.cert_thumbprint IS DISTINCT FROM NEW.cert_thumbprint) THEN
    INSERT INTO auditoria (tabela, operacao, registro_id, dados, usuario)
    VALUES (
      'admins',
      'cert_change',
      NEW.id::text,
      json_build_object(
        'old_thumbprint', OLD.cert_thumbprint,
        'new_thumbprint', NEW.cert_thumbprint,
        'old_method', OLD.auth_method,
        'new_method', NEW.auth_method
      )::text,
      COALESCE(NEW.email, 'system')
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_admin_cert_audit ON admins;
CREATE TRIGGER trg_admin_cert_audit
  BEFORE UPDATE ON admins
  FOR EACH ROW
  EXECUTE FUNCTION audit_cert_change();

Task 4.2 — Add Certificate Functions to auth.py

Add these functions at the bottom of api/auth.py:

# ──────────────────────────────────────────────
# Certificate-Based Authentication (CBA)
# ──────────────────────────────────────────────

from cryptography import x509
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.backends import default_backend
from datetime import datetime, timezone
import hashlib


def carregar_ca_chain(caminho_chain: str) -> list:
    """Load CA chain from PEM file and return list of x509 Certificate objects."""
    with open(caminho_chain, "rb") as f:
        pem_data = f.read()
    certs = []
    for cert in pem_data.split(b"-----END CERTIFICATE-----"):
        cert = cert.strip()
        if cert:
            cert_pem = cert + b"\n-----END CERTIFICATE-----\n"
            try:
                certs.append(x509.load_pem_x509_certificate(cert_pem, default_backend()))
            except Exception:
                continue
    return certs


def extrair_thumbprint(cert_pem: bytes) -> str:
    """Extract SHA-256 thumbprint from a PEM-encoded certificate.
    Returns lowercase hex string (64 chars)."""
    cert = x509.load_pem_x509_certificate(cert_pem, default_backend())
    fingerprint = cert.fingerprint(hashes.SHA256())
    return fingerprint.hex().lower()


def extrair_identidade_cert(cert_pem: bytes) -> dict:
    """Extract identity fields from a client certificate.
    Returns dict with 'cn', 'email', 'subject', 'thumbprint', 'not_after'."""
    cert = x509.load_pem_x509_certificate(cert_pem, default_backend())
    subject = cert.subject

    # Extract CN
    cn_attrs = subject.get_attributes_for_oid(x509.oid.NameOID.COMMON_NAME)
    cn = cn_attrs[0].value if cn_attrs else ""

    # Extract email from subject or SAN extension
    email = ""
    email_attrs = subject.get_attributes_for_oid(x509.oid.NameOID.EMAIL_ADDRESS)
    if email_attrs:
        email = email_attrs[0].value
    else:
        try:
            san = cert.extensions.get_extension_for_class(x509.SubjectAlternativeName)
            emails = san.value.get_values_for_type(x509.RFC822Name)
            if emails:
                email = emails[0]
        except x509.ExtensionNotFound:
            pass

    thumbprint = cert.fingerprint(hashes.SHA256()).hex().lower()

    return {
        "cn": cn,
        "email": email,
        "subject": subject.rfc4514_string(),
        "thumbprint": thumbprint,
        "not_after": cert.not_valid_after_utc.isoformat(),
    }


def validar_certificado(cert_pem: bytes, ca_chain: list) -> dict:
    """Validate a client certificate against the CA chain.
    Returns dict: {'valid': bool, 'error': str|None, 'identity': dict|None}"""
    try:
        cert = x509.load_pem_x509_certificate(cert_pem, default_backend())
    except Exception as e:
        return {"valid": False, "error": f"Invalid certificate format: {e}", "identity": None}

    # 1. Check expiry
    now = datetime.now(timezone.utc)
    if now < cert.not_valid_before_utc:
        return {"valid": False, "error": "Certificate not yet valid", "identity": None}
    if now > cert.not_valid_after_utc:
        return {"valid": False, "error": "Certificate has expired", "identity": None}

    # 2. Check extended key usage includes clientAuth
    try:
        eku = cert.extensions.get_extension_for_class(x509.ExtendedKeyUsage)
        if x509.oid.ExtendedKeyUsageOID.CLIENT_AUTH not in eku.value:
            return {"valid": False, "error": "Certificate not valid for client auth", "identity": None}
    except x509.ExtensionNotFound:
        return {"valid": False, "error": "Missing extendedKeyUsage extension", "identity": None}

    # 3. Verify signature chain
    issuer_found = False
    for ca_cert in ca_chain:
        try:
            # Verify the client cert was signed by this CA cert
            ca_cert.public_key().verify(
                cert.signature,
                cert.tbs_certificate_bytes,
                # Use the signature algorithm from the cert
                cert.signature_hash_algorithm,
            )
            issuer_found = True
            break
        except Exception:
            continue

    if not issuer_found:
        return {"valid": False, "error": "Certificate not signed by trusted CA", "identity": None}

    identity = extrair_identidade_cert(cert_pem)
    return {"valid": True, "error": None, "identity": identity}
💡 Note: The validar_certificado function uses the cryptography library's native signature verification. For RSA certs, use padding.PKCS1v15(). The simplified verify above works for basic chain validation — production should use cryptography.x509.verification (Python 3.10+).

    or direct SSL connection)."""

    # 1. Extract client certificate from request
    # In production (Azure App Service), the cert comes as a header:
    cert_header = request.headers.get("X-ARR-ClientCert", "")

    # In local dev with direct SSL, the cert is in the ASGI scope:
    if not cert_header:
        # Try to get from the TLS transport info
        cert_header = request.headers.get("X-Client-Cert", "")

    if not cert_header:
        raise HTTPException(
            status_code=401,
            detail="No client certificate provided. Ensure your browser sent a certificate."
        )

    # 2. Decode the certificate (Azure sends it as base64-encoded DER)
    import base64
    try:
        # Try base64 decode (Azure App Service format)
        cert_der = base64.b64decode(cert_header)
        # Convert DER to PEM
        from cryptography import x509 as x509_mod
        cert_obj = x509_mod.load_der_x509_certificate(cert_der)
        cert_pem = cert_obj.public_bytes(serialization.Encoding.PEM)
    except Exception:
        # Try as raw PEM
        if "BEGIN CERTIFICATE" in cert_header:
            cert_pem = cert_header.encode()
        else:
            raise HTTPException(status_code=400, detail="Invalid certificate format")

    # 3. Validate the certificate against CA chain
    if not CA_CHAIN:
        raise HTTPException(status_code=503, detail="CBA not configured (CA chain missing)")

    result = validar_certificado(cert_pem, CA_CHAIN)
    if not result["valid"]:
        raise HTTPException(status_code=401, detail=f"Certificate validation failed: {result['error']}")

    # 4. Look up admin by thumbprint
    identity = result["identity"]
    admin = obter_admin_por_cert_thumbprint(identity["thumbprint"])

    if not admin:
        raise HTTPException(
            status_code=401,
            detail="Certificate is valid but not registered to any admin account. "
                   "Please register your certificate first via the profile page."
        )

    # 5. Check auth_method allows cert login
    if admin["auth_method"] == "totp":
        raise HTTPException(
            status_code=403,
            detail="Your account is set to TOTP-only authentication. "
                   "Change to 'cert' or 'both' in your profile settings."
        )

    # 6. Generate JWT token (same as password login)
    token = gerar_token({
        "sub": admin["id"],
        "email": admin["email"],
        "papel": admin["papel"],
        "auth_via": "cert",
    })

    return {
        "access_token": token,
        "token_type": "bearer",
        "admin": {
            "id": admin["id"],
            "email": admin["email"],
            "nome": admin["nome"],
            "auth_via": "cert",

@app.post("/admin/me/cert/register")
async def admin_register_cert(request: Request):
    """Register a client certificate to the currently authenticated admin."""
    # Require existing JWT auth (admin must be logged in via password/TOTP first)
    token = request.headers.get("Authorization", "").replace("Bearer ", "")
    payload = verificar_token(token)
    if not payload:
        raise HTTPException(status_code=401, detail="Invalid or expired token")

    admin_id = payload.get("sub")

    # Get cert from request body (uploaded PEM) or from TLS header
    body = await request.json()
    cert_pem_b64 = body.get("certificate", "")
    if not cert_pem_b64:
        raise HTTPException(status_code=400, detail="No certificate provided in request body")

    import base64
    try:
        cert_pem = base64.b64decode(cert_pem_b64)
    except Exception:
        cert_pem = cert_pem_b64.encode() if isinstance(cert_pem_b64, str) else cert_pem_b64

    # Validate cert
    if not CA_CHAIN:
        raise HTTPException(status_code=503, detail="CBA not configured")

    result = validar_certificado(cert_pem, CA_CHAIN)
    if not result["valid"]:
        raise HTTPException(status_code=400, detail=f"Invalid certificate: {result['error']}")

    identity = result["identity"]

    # Check if thumbprint is already used by another admin
    existing = obter_admin_por_cert_thumbprint(identity["thumbprint"])
    if existing and existing["id"] != admin_id:
        raise HTTPException(status_code=409, detail="This certificate is already registered to another account")

    # Register cert
    registrar_cert_admin(admin_id, identity["thumbprint"], identity["subject"])

    return {
        "message": "Certificate registered successfully",
        "thumbprint": identity["thumbprint"],
        "subject": identity["subject"],
        "expires": identity["not_after"],
    }

async function registerCertificate() {
    const fileInput = document.getElementById('cert-file');
    const msgDiv = document.getElementById('cert-msg');

    if (!fileInput.files || !fileInput.files[0]) {
        showCertMsg('Selecione um arquivo de certificado (.pem)', '#f85149');
        return;
    }

    const file = fileInput.files[0];
    const text = await file.text();

    // Encode as base64 for the API
    const certBase64 = btoa(text);

    try {
        const token = localStorage.getItem('token');
        const response = await fetch(`${API_URL}/admin/me/cert/register`, {
            method: 'POST',
            headers: {
                'Authorization': `Bearer ${token}`,
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({ certificate: certBase64 }),
        });

        const data = await response.json();
        if (response.ok) {
            showCertMsg(`Certificado registrado! Thumbprint: ${data.thumbprint}`, '#3fb950');
            loadCertStatus(); // Refresh UI
        } else {
            showCertMsg(data.detail || 'Erro ao registrar certificado', '#f85149');
        }
    } catch (err) {
        showCertMsg(`Erro: ${err.message}`, '#f85149');
    }
}

async function removeCertificate() {
    if (!confirm('Tem certeza que deseja remover o certificado? O método de autenticação será redefinido para TOTP.')) {
        return;
    }
    try {
        const token = localStorage.getItem('token');
        const response = await fetch(`${API_URL}/admin/me/cert`, {
            method: 'DELETE',
            headers: { 'Authorization': `Bearer ${token}` },
        });
        const data = await response.json();
        if (response.ok) {
            showCertMsg('Certificado removido com sucesso.', '#3fb950');
            loadCertStatus();
        } else {
            showCertMsg(data.detail || 'Erro ao remover certificado', '#f85149');
        }
    } catch (err) {
        showCertMsg(`Erro: ${err.message}`, '#f85149');
    }
}

async function updateAuthMethod() {
    const select = document.getElementById('auth-method-select');
    const method = select.value;

    try {
        const token = localStorage.getItem('token');
        const response = await fetch(`${API_URL}/admin/me/auth-method`, {
            method: 'PUT',

                'Authorization': `Bearer ${token}`,
                'Content-Type': 'application/json',

        }
    } catch (err) {
        showCertMsg(`Erro: ${err.message}`, '#f85149');
    }
}

Certificate Inspection Commands

# View full certificate details
openssl x509 -text -noout -in certs/clients/thiago.pem

# View only subject and issuer
openssl x509 -subject -issuer -noout -in certs/clients/thiago.pem
# subject=C=BR, O=Gen360, CN=Thiago Beier, emailAddress=thiago.beier@gmail.com
# issuer=C=BR, ST=DF, L=Brasilia, O=Gen360, OU=Security, CN=Gen360 Intermediate CA

# Get SHA-256 thumbprint (matches DB cert_thumbprint)
openssl x509 -fingerprint -sha256 -noout -in certs/clients/thiago.pem
# sha256 Fingerprint=A1:B2:C3:D4:E5:F6:...

# Check validity dates
openssl x509 -startdate -enddate -noout -in certs/clients/thiago.pem
# notBefore=Apr  1 00:00:00 2026 GMT
# notAfter=Apr  1 00:00:00 2027 GMT

# Verify certificate chain
openssl verify -CAfile certs/ca-chain.pem certs/clients/thiago.pem
# certs/clients/thiago.pem: OK

# Check extended key usage
openssl x509 -text -noout -in certs/clients/thiago.pem | findstr "Extended"
#             X509v3 Extended Key Usage:
#                 TLS Web Client Authentication

# Check a .p12 file contents
openssl pkcs12 -info -in certs/clients/thiago.p12 -nokeys

# Convert DER to PEM (useful for Azure-provided certs)
openssl x509 -inform DER -in cert.der -out cert.pem

# Test TLS connection and see client cert exchange
openssl s_client -connect localhost:8000 -cert certs/clients/thiago.pem -key certs/clients/thiago.key -CAfile certs/ca-chain.pem
